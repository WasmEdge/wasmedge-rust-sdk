16#[cfg(unix)]
17pub(crate) fn path_to_cstring(path: &Path) -> WasmEdgeResult<CString> {
18    use std::os::unix::ffi::OsStrExt;
19    CString::new(path.as_os_str().as_bytes())
20        .map_err(|err| Box::new(WasmEdgeError::FoundNulByte(err)))
21}
22
23#[cfg(windows)]
24pub(crate) fn path_to_cstring(path: &Path) -> WasmEdgeResult<CString> {
25    match path.to_str() {
26        Some(s) => CString::new(s).map_err(|err| Box::new(WasmEdgeError::FoundNulByte(err))),
27        None => Err(Box::new(WasmEdgeError::WindowsPathConversion(
28            path.to_string_lossy().to_string(),
29        ))),
30    }
31}

48// Checks the result of a `FFI` function.
49pub(crate) fn check(result: WasmEdge_Result) -> WasmEdgeResult<()> {
50    let category = unsafe { ffi::WasmEdge_ResultGetCategory(result) };
51    let code = unsafe {
52        if !WasmEdge_ResultOK(result) {
53            WasmEdge_ResultGetCode(result)
54        } else {
55            0u32
56        }
57    } as ffi::WasmEdge_ErrCode;
58
59    match category {
60        ffi::WasmEdge_ErrCategory_UserLevelError => Err(Box::new(WasmEdgeError::User(code as _))),
61        ffi::WasmEdge_ErrCategory_WASM => gen_runtime_error(code),
62        _ => panic!("Invalid category value: {category}"),
63    }
64}

438impl From<CoreError> for WasmEdge_Result {
439    fn from(val: CoreError) -> WasmEdge_Result {
440        let code = match val {
441            // Common errors
442            CoreError::Common(e) => match e {
443                CoreCommonError::Terminated => ffi::WasmEdge_ErrCode_Terminated,
444                CoreCommonError::RuntimeError => ffi::WasmEdge_ErrCode_RuntimeError,
445                CoreCommonError::CostLimitExceeded => ffi::WasmEdge_ErrCode_CostLimitExceeded,
446                CoreCommonError::WrongVMWorkflow => ffi::WasmEdge_ErrCode_WrongVMWorkflow,
447                CoreCommonError::FuncNotFound => ffi::WasmEdge_ErrCode_FuncNotFound,
448                CoreCommonError::AOTDisabled => ffi::WasmEdge_ErrCode_AOTDisabled,
449                CoreCommonError::Interrupted => ffi::WasmEdge_ErrCode_Interrupted,
450                CoreCommonError::UserDefError => ffi::WasmEdge_ErrCode_UserDefError,
451                CoreCommonError::NotValidated => ffi::WasmEdge_ErrCode_NotValidated,
452                CoreCommonError::NonNullRequired => ffi::WasmEdge_ErrCode_NonNullRequired,
453                CoreCommonError::SetValueToConst => ffi::WasmEdge_ErrCode_SetValueToConst,
454                CoreCommonError::SetValueErrorType => ffi::WasmEdge_ErrCode_SetValueErrorType,
455            },
456
457            // Load phase
458            CoreError::Load(e) => match e {
459                CoreLoadError::IllegalPath => ffi::WasmEdge_ErrCode_IllegalPath,
460                CoreLoadError::ReadError => ffi::WasmEdge_ErrCode_ReadError,
461                CoreLoadError::UnexpectedEnd => ffi::WasmEdge_ErrCode_UnexpectedEnd,
462                CoreLoadError::MalformedMagic => ffi::WasmEdge_ErrCode_MalformedMagic,
463                CoreLoadError::MalformedVersion => ffi::WasmEdge_ErrCode_MalformedVersion,
464                CoreLoadError::MalformedSection => ffi::WasmEdge_ErrCode_MalformedSection,
465                CoreLoadError::SectionSizeMismatch => ffi::WasmEdge_ErrCode_SectionSizeMismatch,
466                CoreLoadError::LengthOutOfBounds => ffi::WasmEdge_ErrCode_LengthOutOfBounds,
467                CoreLoadError::JunkSection => ffi::WasmEdge_ErrCode_JunkSection,

648/// Triggers the WasmEdge AOT compiler tool
649pub fn driver_aot_compiler<I, V>(args: I) -> i32
650where
651    I: IntoIterator<Item = V>,
652    V: AsRef<str>,
653{
654    // create a vector of zero terminated strings
655    let args = args
656        .into_iter()
657        .map(|arg| CString::new(arg.as_ref()).unwrap())
658        .collect::<Vec<CString>>();
659
660    // convert the strings to raw pointers
661    let mut c_args = args
662        .iter()
663        .map(|arg| arg.as_ptr())
664        .collect::<Vec<*const std::os::raw::c_char>>();
665
666    unsafe {
667        ffi::WasmEdge_Driver_Compiler(c_args.len() as std::os::raw::c_int, c_args.as_mut_ptr())
668    }
669}
